-- SP HISTORICO CINTURON
-- -------------------------------------------
DELIMITER $$

CREATE PROCEDURE `sp_generar_hist_cinturon` (in an_cod_estudiante int(10),
                                         in an_cod_cinturon   int(5),
                                         in as_marca varchar(1),
										                     in as_cod_usuario 	  varchar(10))  
BEGIN
  
  -- Declarar variables
  DECLARE li_cuenta int;
  DECLARE li_orden int;
  
  -- Contar Cinturones 
  SELECT COUNT(1) INTO li_cuenta 
    FROM MAE_ESTUDIANTE_CINTURON e 
   WHERE e.N_COD_ESTUDIANTE = an_cod_estudiante
     AND e.N_COD_CINTURON = an_cod_cinturon
     AND e.V_FLAG_ESTADO = '1';
  
  -- Si existe Limpiar
  IF li_cuenta > 0 THEN
    DELETE FROM MAE_ESTUDIANTE_CINTURON WHERE N_COD_ESTUDIANTE = an_cod_estudiante AND N_COD_CINTURON = an_cod_cinturon;
  END IF;

  -- Si está marcado inserta
  IF as_marca = '1' THEN
    -- Orden
    SELECT e.n_orden INTO li_orden
    FROM MAE_CINTURON e 
   WHERE e.N_COD_CINTURON = an_cod_cinturon;

    -- Insertar Cinturon
    IF li_cuenta = 0 THEN
        INSERT INTO MAE_ESTUDIANTE_CINTURON(n_cod_estudiante, n_cod_cinturon, n_orden, v_flag_estado, v_aud_usr_reg, d_aud_fec_reg)
        VALUES (an_cod_estudiante, an_cod_cinturon, li_orden, '1', as_cod_usuario, NOW() );
    END IF;
  END IF;

END$$

-- SP GENERAR USUARIO
-- -------------------------------------------
DELIMITER $$

CREATE PROCEDURE `sp_generar_usuario` (in as_cod_tipouser     varchar(10),
                                       in an_cod_referencia   int(10),
                                       in as_clave   varchar(60),
									   in as_cod_usuario 	  varchar(10))  
BEGIN
  
  -- Declarar variables
  DECLARE li_cuenta int;
  DECLARE ls_nro_doc varchar(40);
  DECLARE ls_datos varchar(100);
  DECLARE ls_email varchar(40); 
  DECLARE ls_foto varchar(150); 
  
  -- Consultar si existe usuario (Evaluarlo)
  SELECT COUNT(1) INTO li_cuenta 
    FROM MAE_USUARIO u
   WHERE u.V_COD_TIPO = as_cod_tipouser
     AND u.N_COD_REFERENCIA = an_cod_referencia;
     
  -- No existe Usuario
  IF li_cuenta = 0 THEN
  	
    -- Capturar datos segun caso
    IF as_cod_tipouser = 'EST_INT' OR as_cod_tipouser = 'EST_EXT' THEN
      SELECT e.V_NRO_DOC, CONCAT_WS(' ', e.V_APE_PATERNO, e.V_APE_MATERNO, e.V_NOMBRES) as datos, e.V_EMAIL, e.V_FOTO 
        INTO ls_nro_doc, ls_datos, ls_email, ls_foto
        FROM MAE_ESTUDIANTE e
       WHERE e.N_COD_ESTUDIANTE = an_cod_referencia;
    ELSE
      SELECT i.V_NRO_DOC, CONCAT_WS(' ', i.V_APE_PATERNO, i.V_APE_MATERNO, i.V_NOMBRES) as datos, i.V_EMAIL, i.V_FOTO -- USER_INST
        INTO ls_nro_doc, ls_datos, ls_email, ls_foto
        FROM MAE_INSTRUCTOR i
       WHERE i.N_COD_INSTRUCTOR = an_cod_referencia;
    END IF;

  	-- Insertar Usuario
      INSERT INTO MAE_USUARIO(v_cod_user, v_nombres, v_email, v_clave, v_foto, d_fec_alta, v_cod_tipo, n_cod_referencia, v_observacion, v_flag_estado, v_aud_usr_reg, d_aud_fec_reg)
      VALUES (ls_nro_doc, ls_datos, ls_email, as_clave, ls_foto, CURDATE(), as_cod_tipouser, an_cod_referencia, 'Generado automaticamente', '1', as_cod_usuario, NOW() );
   END IF;
  
END$$

-- SP HORARIO - PROGRAMACIÓN
-- -------------------------------------------
DELIMITER $$

CREATE PROCEDURE `sp_generar_progclases` (in an_cod_horario int(10),
											  in as_cod_usuario varchar(10))
BEGIN

  -- Declarar variables
  DECLARE li_sesion int DEFAULT 0; 
  DECLARE ld_fec_inicio date;
  DECLARE ld_fec_clase  date;
  DECLARE ldt_hora_inicio time;
  DECLARE li_horas int;
  DECLARE li_sesiones int;
  DECLARE li_cod_instructor int; 
  DECLARE ls_fg_1 varchar(1);
  DECLARE ls_fg_2 varchar(1);
  DECLARE ls_fg_3 varchar(1);
  DECLARE ls_fg_4 varchar(1);
  DECLARE ls_fg_5 varchar(1);
  DECLARE ls_fg_6 varchar(1);
  DECLARE ls_fg_7 varchar(1);
  DECLARE li_diaSem int DEFAULT 0; 
  DECLARE li_dia_1 int DEFAULT 0; 
  DECLARE li_dia_2 int DEFAULT 0; 
  DECLARE li_dia_3 int DEFAULT 0; 
  DECLARE li_dia_4 int DEFAULT 0; 
  DECLARE li_dia_5 int DEFAULT 0; 
  DECLARE li_dia_6 int DEFAULT 0;
  DECLARE li_dia_7 int DEFAULT 0;   

  -- Obtener datos de horario
  SELECT h.D_FEC_INICIO, h.D_HORA_INICIO, h.N_HORAS, h.N_SESIONES, N_COD_INSTRUCTOR,
         h.V_FLAG_DIA_1, h.V_FLAG_DIA_2, h.V_FLAG_DIA_3, 
         h.V_FLAG_DIA_4, h.V_FLAG_DIA_5, h.V_FLAG_DIA_6, h.V_FLAG_DIA_7
    INTO ld_fec_inicio, ldt_hora_inicio, li_horas, li_sesiones, li_cod_instructor,
         ls_fg_1, ls_fg_2, ls_fg_3, ls_fg_4, ls_fg_5, ls_fg_6, ls_fg_7
    FROM MOV_HORARIO h
   WHERE h.N_COD_HORARIO = an_cod_horario;
  
  -- Capturar Días Programados
  IF (ls_fg_1 = '1') THEN
    SET li_dia_1 = 1;
  END IF;
  IF (ls_fg_2 = '1') THEN
    SET li_dia_2 = 2;
  END IF;
  IF (ls_fg_3 = '1') THEN
    SET li_dia_3 = 3;
  END IF;
  IF (ls_fg_4 = '1') THEN
    SET li_dia_4 = 4;
  END IF;
  IF (ls_fg_5 = '1') THEN
    SET li_dia_5 = 5;
  END IF;
  IF (ls_fg_6 = '1') THEN
    SET li_dia_6 = 6;
  END IF;
  IF (ls_fg_7 = '1') THEN
    SET li_dia_7 = 7;
  END IF;

  -- Inicializar
  SET ld_fec_clase = ld_fec_inicio;

  -- Bucle de Sesiones
  WHILE (li_sesiones > li_sesion) DO
    
    -- Capturar Día
    SELECT DAYOFWEEK(ld_fec_clase) INTO li_diaSem;

    -- Si es Día
    IF ( li_diaSem = li_dia_1 OR li_diaSem = li_dia_2 OR li_diaSem = li_dia_3 
       OR li_diaSem = li_dia_4 OR li_diaSem = li_dia_5 OR li_diaSem = li_dia_6 
       OR li_diaSem = li_dia_7 ) THEN
      
      -- Incrementar Sesión
      SET li_sesion = li_sesion + 1;

      -- Insertar Fila Clase 
      INSERT INTO MOV_HORARIO_PROG(N_COD_HORARIO, N_CLASE, N_COD_INSTRUCTOR, D_FEC_PROG, D_HORA_PROG, N_HORAS, V_OBSERVACION, V_FLAG_ESTADO, V_AUD_USR_REG, D_AUD_FEC_REG)
      VALUES (an_cod_horario, li_sesion, li_cod_instructor, ld_fec_clase, ldt_hora_inicio, li_horas, 'Generado automaticamente', '1', as_cod_usuario, NOW() );

    END IF;

    -- Incrementar Fecha
    SET ld_fec_clase = ld_fec_clase + 1;    

  END WHILE;

END$$
